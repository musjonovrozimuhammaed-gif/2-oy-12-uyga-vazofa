#include <stdio.h>
void factorial(int n, long long *result) {
    if (n <= 1) {
        return;
    }
    *result *= n;
    factorial(n - 1, result);
}
int main() {
    int n;
    long long result = 1;
    printf("N ni kiriting: ");
    scanf("%d", &n);
    if (n < 0) {
        printf("Manfiy son uchun faktorial aniqlanmagan.\n");
    } else {
        factorial(n, &result);
        printf("%d! = %lld\n", n, result);
    }
    return 0;
}
